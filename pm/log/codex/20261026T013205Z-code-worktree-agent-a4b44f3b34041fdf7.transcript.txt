+/** A ladder of `levels` per side around `mid`, each `step` apart, `size` on every level. */
+export function ladderOf(mid: number, levels: number, step: number, size: number, ts = 0): L2Ladder {
+  const bids: [number, number][] = [];
+  const asks: [number, number][] = [];
+  for (let level = 1; level <= levels; level += 1) {
+    bids.push([Math.round((mid - level * step) * 100) / 100, size]);
+    asks.push([Math.round((mid + level * step) * 100) / 100, size]);
+  }
+  return { ts, depth: levels, bids, asks };
+}

+/** A ladder with one level inflated to `wallSize`: a resting wall at `wallPrice`. */
+export function ladderWithWall(base: L2Ladder, side: "bid" | "ask", wallPrice: number, wallSize: number): L2Ladder {
+  const bump = (levels: ReadonlyArray<readonly [number, number]>) => levels.map(([price, size]) => (price === wallPrice ? ([price, wallSize] as const) : ([price, size] as const)));
+  return { ...base, bids: side === "bid" ? bump(base.bids) : base.bids, asks: side === "ask" ? bump(base.asks) : base.asks };

+/** Linear congruential generator: the same seed gives the same fixture on every machine. */
+export function seeded(seed: number): () => number {
+  let state = seed >>> 0 || 1;
+  return () => {

+    return state / 4294967296;
+  };

+export function barFrame(

+  spec: BarSpec,
+  identity: FixtureIdentity = BINANCE_PERP,
+  intervalMs = MINUTE_MS,
+  closed = true,
+): BarFrame {
+  const volume = round(spec.volume, 4);
+  const frame: Record<string, unknown> = {
+    type: "bar",
+    ...identity,
+    interval_ms: intervalMs,

+    low: round(Math.min(spec.low, spec.open, spec.close)),
+    close: round(spec.close),

+function barPath(bar: BarFrame, points: number): number[] {
+  const up = bar.close >= bar.open;
+  const anchors = up ? [bar.open, bar.low, bar.high, bar.close] : [bar.open, bar.high, bar.low, bar.close];
+  const out: number[] = [];
+  for (let index = 0; index < points; index += 1) {
+    const t = ((index + 1) / (points + 1)) * (anchors.length - 1);
+    const segment = Math.min(anchors.length - 2, Math.floor(t));
+    const within = t - segment;
+    out.push(anchors[segment] + (anchors[segment + 1] - anchors[segment]) * within);
+  }
+  return out;
+}
+
+/**
+ * The forming revisions of one closed bar: `revisions` frames that walk the
+ * bar's path from its open through its wicks to its close, each `closed:
+ * false`, followed by the closed bar itself. The chart appends the first one
+ * and replaces the edge with each later one, which is how a live bar really
+ * reaches an indicator many times before it is committed.
+ */
+export function formingRevisions(bar: BarFrame, revisions: number): BarFrame[] {
+  if (revisions <= 0) return [bar];
+  const path = barPath(bar, revisions);
+  const out: BarFrame[] = [];
+  let high = bar.open;
+  let low = bar.open;
+  const sided = bar.buy_volume !== undefined && bar.buy_volume !== null;
+  for (let index = 0; index < revisions; index += 1) {
+    const close = path[index];
+    high = Math.max(high, close);
+    low = Math.min(low, close);
+    const share = (index + 1) / (revisions + 1);
+    const volume = round(bar.volume * share, 4);
+    const partial: Record<string, unknown> = { ...bar, high: round(high), low: round(low), close: round(close), volume, closed: false };
+    if (bar.trade_count !== null) partial.trade_count = Math.max(1, Math.round(bar.trade_count * share));
+    if (sided) {
+      const buy = round((bar.buy_volume ?? 0) * share, 4);
+      partial.buy_volume = buy;
+      if (bar.sell_volume !== undefined) partial.sell_volume = round(volume - buy, 4);
+    }
+    out.push(partial as BarFrame);
+  }
+  out.push(bar);
+  return out;
+}
+
+/** Every bar arriving live: `revisionsPerBar` forming revisions, then the closed bar. */
+export function liveArrivals(
+  bars: readonly BarFrame[],
+  revisionsPerBar: number,
+  tradesFor?: (bar: BarFrame, revision: number, revisions: number) => Pick<Arrival, "trades" | "tradeAvailability">,
+): Arrival[] {
+  const out: Arrival[] = [];
+  for (const bar of bars) {
+    const frames = formingRevisions(bar, revisionsPerBar);
+    frames.forEach((frame, revision) => {
+      out.push({ bar: frame, ...(tradesFor ? tradesFor(frame, revision, frames.length) : {}) });
+    });
+  }
+  return out;
+}

+function copyOutputs(outputs: Outputs | null): Outputs {
+  if (outputs === null) return {};
+  const copy: Record<string, (number | null)[]> = {};
+  for (const [key, values] of Object.entries(outputs)) copy[key] = [...values];
+  return copy;

+export function captureOf(runtime: IndicatorRuntime, id: string, hasLines: boolean): Capture {
+  const outputs = runtime.getOutput(id);
+  const markers = runtime.getMarkers(id);
+  return {
+    outputs: copyOutputs(outputs),
+    markers: markers === null ? [] : markers.map((call) => ({ ...call })),
+    faulted: hasLines ? outputs === null : markers === null,
+  };

+export function openLiveSession<S, T>(
+  def: RuntimeIndicatorDef<S, T>,
+  settings: T,
+  extras: LiveExtras = {},
+  warmup: readonly BarFrame[] = [],
+): LiveSession<S, T> {
+  const runtime = createIndicatorRuntime([def as RuntimeIndicatorDef<unknown, unknown>]);
+  const bars: BarFrame[] = [...warmup];
+  const mutations: ChartBarMutation[] = [];
+  let discarded = 0;
+  if (extras.ticks !== undefined) runtime.setTicks(extras.ticks, []);
+  if (extras.l2 !== undefined) runtime.setL2(extras.l2, []);
+  if (extras.liquidity !== undefined) runtime.setLiquidity(extras.liquidity, []);
+  runtime.setActive(new Set([def.id]), { [def.id]: settings }, bars);
+
+  const session: LiveSession<S, T> = {
+    def,
+    runtime,
+    bars,
+    mutations,
+    get discarded() {
+      return discarded;
+    },
+    setTicks(ticks) {
+      runtime.setTicks(ticks, bars);
+    },
+    setL2(l2) {
+      runtime.setL2(l2, bars);
+    },
+    updateSettings(next) {
+      runtime.updateSettings(def.id, next, bars);
+    },
+    dispatch(arrival) {
+      arrival.act?.(session);
+      const mutation = mergeChartBar(bars, arrival.bar);
+      mutations.push(mutation);
+      if (mutation === "REJECTED_REOPEN") return mutation;
+      let removed = 0;
+      if (bars.length > MAX_RETAINED_SOURCE_BARS) {
+        removed = bars.length - MAX_RETAINED_SOURCE_BARS;
+        bars.splice(0, removed);
+        discarded += removed;
+      }
+      runtime.onBar(arrival.bar, mutation, bars, arrival.trades, arrival.tradeAvailability);
+      if (removed > 0 && mutation === "APPENDED") runtime.discardLeading(removed, bars[0]?.ts_open);
+      return mutation;
+    },
+    capture() {
+      return captureOf(runtime, def.id, def.lines.length > 0);
+    },
+    close() {
+      runtime.setActive(new Set(), {}, []);
+    },
+  };
+  return session;

+export function valuesEqual(left: number | null | undefined, right: number | null | undefined, tolerance: number): boolean {
+  if (left === right) return true;
+  if (left === null || left === undefined || right === null || right === undefined) return false;
+  if (Number.isNaN(left) || Number.isNaN(right)) return false;
+  return Math.abs(left - right) <= tolerance;

+/** Every position where two output maps disagree; keys present on one side only count as disagreements. */
+export function diffOutputs(left: Outputs, right: Outputs, tolerance = 0, upToIndex = Infinity): OutputDiff[] {
+  const diffs: OutputDiff[] = [];
+  const keys = new Set([...Object.keys(left), ...Object.keys(right)]);
+  for (const key of keys) {
+    const a = left[key] ?? [];
+    const b = right[key] ?? [];
+    const length = Math.min(Math.max(a.length, b.length), upToIndex + 1);
+    for (let index = 0; index < length; index += 1) {
+      if (!valuesEqual(a[index], b[index], tolerance)) diffs.push({ index, key, left: a[index], right: b[index] });
+    }
+  }
+  return diffs;
+}

+/** Multiset difference of two marker lists, order-insensitive. */
+export function diffMarkers(left: readonly MarkerCall[], right: readonly MarkerCall[], upToTime = Infinity): MarkerDiff {
+  const counts = new Map<string, number>();
+  const byKey = new Map<string, MarkerCall>();
+  for (const call of left) {
+    if (call.time > upToTime) continue;
+    const key = markerKey(call);
+    counts.set(key, (counts.get(key) ?? 0) + 1);
+    byKey.set(key, call);
+  }
+  const rightOnly: MarkerCall[] = [];
+  for (const call of right) {
+    if (call.time > upToTime) continue;
+    const key = markerKey(call);
+    const count = counts.get(key) ?? 0;
+    if (count === 0) rightOnly.push(call);
+    else counts.set(key, count - 1);
+  }
+  const leftOnly: MarkerCall[] = [];
+  for (const [key, count] of counts) {
+    const call = byKey.get(key)!;
+    for (let index = 0; index < count; index += 1) leftOnly.push(call);
+  }
+  return { leftOnly, rightOnly };

+export function percentilesOf(samples: readonly number[]): Percentiles {
+  if (samples.length === 0) return { count: 0, p50: 0, p90: 0, p99: 0, max: 0, mean: 0, totalMs: 0 };
+  const sorted = [...samples].sort((a, b) => a - b);
+  const at = (quantile: number) => sorted[Math.min(sorted.length - 1, Math.max(0, Math.ceil(quantile * sorted.length) - 1))];
+  const totalMs = sorted.reduce((sum, value) => sum + value, 0);
+  return { count: sorted.length, p50: at(0.5), p90: at(0.9), p99: at(0.99), max: sorted[sorted.length - 1], mean: totalMs / sorted.length, totalMs };

+  return tape;
+}

+export function ticksOf(tape: AggressiveTape): IndicatorTicks {
+  const ticks = indicatorTicksOf(tape);
+  if (ticks === null) throw new Error("a tape always yields a ticks view");
+  return ticks;
+}

+export interface CandleOfOptions {
+  identity?: FixtureIdentity;
+  intervalMs?: number;
+  /** `null` withholds the print count, the one candle field that would tell a whale from a crowd. */
+  tradeCount?: number | null;
+  closed?: boolean;
+}
+
+/** The candle these prints make: OHLC from the price path, volume and sides summed, count as given. */
+export function candleOf(prints: readonly TradeFrame[], ts_open: number, options: CandleOfOptions = {}): BarFrame {
+  if (prints.length === 0) throw new Error("candleOf: a candle needs at least one print");
+  const sorted = [...prints].sort((a, b) => a.ts_exchange - b.ts_exchange);
+  let volume = 0;
+  let buy = 0;
+  let high = -Infinity;
+  let low = Infinity;
+  for (const print of sorted) {
+    volume += print.size;
+    if (print.side === "buy") buy += print.size;
+    high = Math.max(high, print.price);
+    low = Math.min(low, print.price);
+  }
+  const frame = barFrame(
+    ts_open,
+    {
+      open: sorted[0].price,
+      close: sorted[sorted.length - 1].price,
+      high,
+      low,
+      volume,
+      buyShare: volume > 0 ? buy / volume : 0,
+      tradeCount: options.tradeCount === undefined ? prints.length : options.tradeCount,
+    },
+    options.identity ?? BINANCE_PERP,
+    options.intervalMs ?? MINUTE_MS,
+    options.closed ?? true,
+  );
+  return frame;
+}

+export interface WhaleVersusCrowdOptions {
+  ts_open?: number;
+  intervalMs?: number;
+  price?: number;
+  /** Size of the one whale print; the crowd's prints sum to exactly this. */
+  whaleSize?: number;
+  crowdCount?: number;
+  side?: Side;
+  identity?: FixtureIdentity;
+  /**
+   * Shared small prints along a price path that both tapes contain, so the
+   * candle has a body and wicks. The whale and the crowd then print at `price`
+   * at the end of the bar.
+   */
+  pathPrints?: number;
+  pathSize?: number;
+  seed?: number;
+}
+
+export interface WhaleVersusCrowd {
+  /** The one large print (plus any shared path prints). */
+  whale: TradeFrame[];
+  /** `crowdCount` small prints with the same total (plus the same path prints). */
+  crowd: TradeFrame[];
+  /** The candle both tapes make, with the print count withheld (`trade_count: null`). */
+  candle: BarFrame;
+  /** The same candle with its honest counts: the only field on which the two tapes differ. */
+  whaleCandle: BarFrame;
+  crowdCandle: BarFrame;
+  whaleTape: AggressiveTape;
+  crowdTape: AggressiveTape;
+}
+
+export function whaleVersusCrowd(options: WhaleVersusCrowdOptions = {}): WhaleVersusCrowd {
+  const ts_open = options.ts_open ?? 1_780_000_000_000;
+  const intervalMs = options.intervalMs ?? MINUTE_MS;
+  const price = options.price ?? 60_000;
+  const whaleSize = options.whaleSize ?? 50;
+  const crowdCount = options.crowdCount ?? 100;
+  const side = options.side ?? "buy";
+  const identity = options.identity ?? BINANCE_PERP;
+  const pathPrints = options.pathPrints ?? 0;
+  const pathSize = options.pathSize ?? 0.2;
+  const random = seeded(options.seed ?? 11);
+
+  const path: TradeFrame[] = [];
+  for (let index = 0; index < pathPrints; index += 1) {
+    const ts = ts_open + Math.floor(((index + 1) / (pathPrints + 2)) * intervalMs * 0.8);
+    const wander = price + (random() - 0.5) * price * 0.001;
+    path.push(tradeFrame(ts, Math.round(wander * 100) / 100, pathSize, random() < 0.5 ? "buy" : "sell", identity, `path-${index}`));
+  }
+  const burstStart = ts_open + Math.floor(intervalMs * 0.85);
+  const whale = [...path, tradeFrame(burstStart, price, whaleSize, side, identity, "whale")];
+  const crowd = [...path];
+  const small = Math.round((whaleSize / crowdCount) * 1e8) / 1e8;
+  let remaining = whaleSize;
+  for (let index = 0; index < crowdCount; index += 1) {
+    const size = index === crowdCount - 1 ? Math.round(remaining * 1e8) / 1e8 : small;
+    remaining -= size;
+    crowd.push(tradeFrame(burstStart + Math.floor((index * intervalMs * 0.1) / crowdCount), price, size, side, identity, `crowd-${index}`));
+  }
+  const candleOptions = { identity, intervalMs };
+  return {
+    whale,
+    crowd,
+    candle: candleOf(whale, ts_open, { ...candleOptions, tradeCount: null }),
+    whaleCandle: candleOf(whale, ts_open, candleOptions),
+    crowdCandle: candleOf(crowd, ts_open, candleOptions),
+    whaleTape: tapeOf(whale, ts_open),
+    crowdTape: tapeOf(crowd, ts_open),
+  };
+}

+export interface BusyTapeOptions {
+  printsPerBar?: number;
+  /** Fraction of prints that are "large" (ten times the small size). */
+  largeShare?: number;
+  /** Fraction of prints that are whales (`whaleMultiple` times the small size); rare by default. */
+  whaleShare?: number;
+  whaleMultiple?: number;
+  smallSize?: number;
+  identity?: FixtureIdentity;
+  seed?: number;
+}
+
+/** Prints for every bar of `bars`, inside each bar's window and range, with rare whales: the busy instrument the marker budget is measured on. */
+export function busyTapeFor(bars: readonly BarFrame[], options: BusyTapeOptions = {}): Map<number, TradeFrame[]> {
+  const printsPerBar = options.printsPerBar ?? 60;
+  const largeShare = options.largeShare ?? 0.02;
+  const whaleShare = options.whaleShare ?? 0.002;
+  const whaleMultiple = options.whaleMultiple ?? 100;
+  const smallSize = options.smallSize ?? 0.1;
+  const identity = options.identity ?? BINANCE_PERP;
+  const random = seeded(options.seed ?? 5);
+  const out = new Map<number, TradeFrame[]>();
+  for (const bar of bars) {
+    const prints: TradeFrame[] = [];
+    for (let index = 0; index < printsPerBar; index += 1) {
+      const ts = bar.ts_open + Math.floor(((index + 0.5) / printsPerBar) * bar.interval_ms);
+      const price = Math.round((bar.low + random() * (bar.high - bar.low)) * 100) / 100;
+      const draw = random();
+      const size = draw < whaleShare ? smallSize * whaleMultiple : draw < whaleShare + largeShare ? smallSize * 10 : smallSize * (0.5 + random());
+      prints.push(tradeFrame(ts, price, Math.round(size * 1e6) / 1e6, random() < 0.5 ? "buy" : "sell", identity, `${bar.ts_open}-${index}`));
+    }
+    out.set(bar.ts_open, prints);
+  }
+  return out;
+}

+import type { BarFrame } from "../../../types/frames";
+import type { MarkerCall } from "../draw/markers";
+import type { RuntimeIndicatorDef } from "../indicatorTypes";
+import { diffMarkers, diffOutputs, replayCapture, type LiveExtras, type OutputDiff } from "./liveDrive";

+/**
+ * Protocol check 1, no future data. A bar's committed output must be the same
+ * whether or not later bars exist. The series is replayed once in full and
+ * then truncated at every index (or at `indices`); the truncated replay's
+ * outputs and markers up to the cut must equal the full replay's. Anything
+ * that differs is a value that depended on bars after its own close -- a
+ * repaint, a centred window, a `barCount`-anchored lookahead.
+ */
+
+export interface TruncationOptions {
+  /** Indices to truncate at; every index by default (`sampledIndices` cheapens a long series). */
+  indices?: readonly number[];
+  /** Absolute tolerance for numeric outputs; 0 (exact) by default. */
+  tolerance?: number;
+  extras?: LiveExtras;
+  /** Compare markers too (default true). */
+  markers?: boolean;
+}
+
+export interface MarkerViolation {
+  truncatedAt: number;
+  /** Markers the full replay placed at or before the cut that the truncated replay did not. */
+  missing: MarkerCall[];
+  /** Markers the truncated replay placed that the full replay did not. */
+  extra: MarkerCall[];
+}
+
+export interface TruncationReport {
+  ok: boolean;
+  bars: number;
+  checked: number;
+  /** `[truncatedAt, diff]`: at which cut each output disagreement showed. */
+  violations: { truncatedAt: number; diff: OutputDiff }[];
+  markerViolations: MarkerViolation[];
+  faulted: boolean;
+}
+
+export function checkTruncationInvariance<S, T>(
+  def: RuntimeIndicatorDef<S, T>,
+  bars: readonly BarFrame[],
+  settings: T,
+  options: TruncationOptions = {},
+): TruncationReport {
+  const tolerance = options.tolerance ?? 0;
+  const compareMarkers = options.markers ?? true;
+  const indices = options.indices ?? bars.map((_, index) => index);
+  const full = replayCapture(def, bars, settings, options.extras);
+  const violations: TruncationReport["violations"] = [];
+  const markerViolations: MarkerViolation[] = [];
+  let faulted = full.faulted;
+  let checked = 0;
+  for (const cut of indices) {
+    if (cut < 0 || cut >= bars.length) continue;
+    checked += 1;
+    const truncated = replayCapture(def, bars.slice(0, cut + 1), settings, options.extras);
+    faulted = faulted || truncated.faulted;
+    for (const diff of diffOutputs(full.outputs, truncated.outputs, tolerance, cut)) violations.push({ truncatedAt: cut, diff });
+    if (compareMarkers) {
+      const { leftOnly, rightOnly } = diffMarkers(full.markers, truncated.markers, bars[cut].ts_open);
+      if (leftOnly.length > 0 || rightOnly.length > 0) markerViolations.push({ truncatedAt: cut, missing: leftOnly, extra: rightOnly });
+    }
+  }
+  return {
+    ok: !faulted && violations.length === 0 && markerViolations.length === 0,
+    bars: bars.length,
+    checked,
+    violations,
+    markerViolations,
+    faulted,
+  };
+}
+
+/** One line per distinct offending key, for the evidence file. */
+export function summarizeTruncation(report: TruncationReport): string {
+  if (report.ok) return `truncation invariance: PASS (${report.checked} cuts over ${report.bars} bars)`;
+  const lines = [`truncation invariance: FAIL (${report.checked} cuts over ${report.bars} bars)`];
+  if (report.faulted) lines.push("  the indicator faulted during a replay");
+  const byKey = new Map<string, { count: number; first: TruncationReport["violations"][number] }>();
+  for (const violation of report.violations) {
+    const entry = byKey.get(violation.diff.key);
+    if (entry) entry.count += 1;
+    else byKey.set(violation.diff.key, { count: 1, first: violation });
+  }
+  for (const [key, { count, first }] of byKey) {
+    lines.push(
+      `  ${key}: ${count} disagreement(s); first at bar ${first.diff.index} when truncated at ${first.truncatedAt}: full=${String(first.diff.left)} truncated=${String(first.diff.right)}`,
+    );
+  }
+  for (const violation of report.markerViolations.slice(0, 5)) {
+    lines.push(`  markers truncated at ${violation.truncatedAt}: ${violation.missing.length} missing, ${violation.extra.length} extra`);
+  }
+  if (report.markerViolations.length > 5) lines.push(`  ... ${report.markerViolations.length - 5} more marker disagreement(s)`);
+  return lines.join("\n");
+}

 * does: bars are classified with `mergeChartBar`, the retained array is capped
 * at `MAX_RETAINED_SOURCE_BARS` with `discardLeading` on the rollover, and the

export interface LiveExtras {
  ticks?: IndicatorTicks | null;
  l2?: IndicatorL2 | null;
  liquidity?: IndicatorLiquidity | null;
}

export type Outputs = Readonly<Record<string, readonly (number | null)[]>>;

export interface Capture {
  outputs: Outputs;
  markers: readonly MarkerCall[];
  /** Null when the runtime latched the indicator as faulted. */
  faulted: boolean;
}

export interface LiveSession<S, T> extends ArrivalSession {
  readonly def: RuntimeIndicatorDef<S, T>;
  readonly runtime: IndicatorRuntime;
  readonly bars: BarFrame[];
  readonly mutations: ChartBarMutation[];
  /** How many bars the rollover has trimmed off the front so far. */
  readonly discarded: number;
  dispatch(arrival: Arrival): ChartBarMutation;
  capture(): Capture;
  /** Removes the indicator so its draw handles leave the registry. */
  close(): void;
}

function copyOutputs(outputs: Outputs | null): Outputs {
  if (outputs === null) return {};
  const copy: Record<string, (number | null)[]> = {};
  for (const [key, values] of Object.entries(outputs)) copy[key] = [...values];
  return copy;
}

export function captureOf(runtime: IndicatorRuntime, id: string, hasLines: boolean): Capture {
  const outputs = runtime.getOutput(id);
  const markers = runtime.getMarkers(id);
  return {
    outputs: copyOutputs(outputs),
    markers: markers === null ? [] : markers.map((call) => ({ ...call })),
    faulted: hasLines ? outputs === null : markers === null,
  };
}

export function openLiveSession<S, T>(
  def: RuntimeIndicatorDef<S, T>,
  settings: T,
  extras: LiveExtras = {},
  warmup: readonly BarFrame[] = [],
): LiveSession<S, T> {
  const runtime = createIndicatorRuntime([def as RuntimeIndicatorDef<unknown, unknown>]);
  const bars: BarFrame[] = [...warmup];
  const mutations: ChartBarMutation[] = [];
  let discarded = 0;
  if (extras.ticks !== undefined) runtime.setTicks(extras.ticks, []);
  if (extras.l2 !== undefined) runtime.setL2(extras.l2, []);
  if (extras.liquidity !== undefined) runtime.setLiquidity(extras.liquidity, []);
  runtime.setActive(new Set([def.id]), { [def.id]: settings }, bars);

  const session: LiveSession<S, T> = {
    def,
    runtime,
    bars,
    mutations,
    get discarded() {
      return discarded;
    },
    setTicks(ticks) {
      runtime.setTicks(ticks, bars);
    },
    setL2(l2) {
      runtime.setL2(l2, bars);
    },
    updateSettings(next) {
      runtime.updateSettings(def.id, next, bars);
    },
    dispatch(arrival) {
      arrival.act?.(session);
      const mutation = mergeChartBar(bars, arrival.bar);
      mutations.push(mutation);
      if (mutation === "REJECTED_REOPEN") return mutation;
      let removed = 0;
      if (bars.length > MAX_RETAINED_SOURCE_BARS) {
        removed = bars.length - MAX_RETAINED_SOURCE_BARS;
        bars.splice(0, removed);
        discarded += removed;
      }
      runtime.onBar(arrival.bar, mutation, bars, arrival.trades, arrival.tradeAvailability);
      if (removed > 0 && mutation === "APPENDED") runtime.discardLeading(removed, bars[0]?.ts_open);
      return mutation;
    },
    capture() {
      return captureOf(runtime, def.id, def.lines.length > 0);
    },
    close() {
      runtime.setActive(new Set(), {}, []);
    },
  };
  return session;
}

  mutations: ChartBarMutation[];

/** Feeds `arrivals` through a fresh session and returns what the runtime holds at the end. */
export function driveLive<S, T>(
  def: RuntimeIndicatorDef<S, T>,
  arrivals: readonly Arrival[],
  settings: T,
  extras: LiveExtras = {},
  warmup: readonly BarFrame[] = [],
): LiveResult {
  const session = openLiveSession(def, settings, extras, warmup);
  try {
    for (const arrival of arrivals) session.dispatch(arrival);
    return { ...session.capture(), bars: [...session.bars], mutations: [...session.mutations] };
  } finally {
    session.close();
  }
}

/** A clean replay of `bars` through the runtime (`onSnapshot`), the way activation or a reconnect replays. */
export function replayCapture<S, T>(
  def: RuntimeIndicatorDef<S, T>,
  bars: readonly BarFrame[],
  settings: T,
  extras: LiveExtras = {},
): Capture {
  const session = openLiveSession(def, settings, extras, []);
  try {
    session.runtime.onSnapshot(bars);
    return session.capture();
  } finally {
    session.close();
  }
}

export function valuesEqual(left: number | null | undefined, right: number | null | undefined, tolerance: number): boolean {
  if (left === right) return true;
  if (left === null || left === undefined || right === null || right === undefined) return false;
  if (Number.isNaN(left) || Number.isNaN(right)) return false;
  return Math.abs(left - right) <= tolerance;
}

/** Every position where two output maps disagree; keys present on one side only count as disagreements. */
export function diffOutputs(left: Outputs, right: Outputs, tolerance = 0, upToIndex = Infinity): OutputDiff[] {
  const diffs: OutputDiff[] = [];
  const keys = new Set([...Object.keys(left), ...Object.keys(right)]);
  for (const key of keys) {
    const a = left[key] ?? [];
    const b = right[key] ?? [];
    const length = Math.min(Math.max(a.length, b.length), upToIndex + 1);
    for (let index = 0; index < length; index += 1) {
      if (!valuesEqual(a[index], b[index], tolerance)) diffs.push({ index, key, left: a[index], right: b[index] });
    }
  }
  return diffs;
}

export function markerKey(call: MarkerCall): string {
  return JSON.stringify([call.time, call.shape, call.position, call.color, call.text ?? null, call.size ?? null, call.price ?? null]);
}

export interface MarkerDiff {
  leftOnly: MarkerCall[];
  rightOnly: MarkerCall[];
}

/** Multiset difference of two marker lists, order-insensitive. */
export function diffMarkers(left: readonly MarkerCall[], right: readonly MarkerCall[], upToTime = Infinity): MarkerDiff {
  const counts = new Map<string, number>();
  const byKey = new Map<string, MarkerCall>();
  for (const call of left) {
    if (call.time > upToTime) continue;
    const key = markerKey(call);
    counts.set(key, (counts.get(key) ?? 0) + 1);
    byKey.set(key, call);
  }
  const rightOnly: MarkerCall[] = [];
  for (const call of right) {
    if (call.time > upToTime) continue;
    const key = markerKey(call);
    const count = counts.get(key) ?? 0;
    if (count === 0) rightOnly.push(call);
    else counts.set(key, count - 1);
  }
  const leftOnly: MarkerCall[] = [];
  for (const [key, count] of counts) {
    const call = byKey.get(key)!;
    for (let index = 0; index < count; index += 1) leftOnly.push(call);
  }
  return { leftOnly, rightOnly };
}

export function percentilesOf(samples: readonly number[]): Percentiles {
  if (samples.length === 0) return { count: 0, p50: 0, p90: 0, p99: 0, max: 0, mean: 0, totalMs: 0 };
  const sorted = [...samples].sort((a, b) => a - b);
  const at = (quantile: number) => sorted[Math.min(sorted.length - 1, Math.max(0, Math.ceil(quantile * sorted.length) - 1))];

export function heapUsed(): { bytes: number; gcExposed: boolean } {
  const gc = (globalThis as GcGlobal).gc;

      ...last,

      close,
      volume,
      buy_volume: Math.round(volume * (0.3 + random() * 0.4) * 100) / 100,
      closed: true,
    } as BarFrame;
    session.dispatch({ bar });
    last = bar;
  }

  const capture = session.capture();
  const oldest = session.bars[0]?.ts_open ?? -Infinity;

export function barFrame(

function barPath(bar: BarFrame, points: number): number[] {
  const up = bar.close >= bar.open;
  const anchors = up ? [bar.open, bar.low, bar.high, bar.close] : [bar.open, bar.high, bar.low, bar.close];
  const out: number[] = [];
  for (let index = 0; index < points; index += 1) {
    const t = ((index + 1) / (points + 1)) * (anchors.length - 1);
    const segment = Math.min(anchors.length - 2, Math.floor(t));
    const within = t - segment;
    out.push(anchors[segment] + (anchors[segment + 1] - anchors[segment]) * within);
  }
  return out;
}

/**
 * The forming revisions of one closed bar: `revisions` frames that walk the
 * bar's path from its open through its wicks to its close, each `closed:
 * false`, followed by the closed bar itself. The chart appends the first one
 * and replaces the edge with each later one, which is how a live bar really
 * reaches an indicator many times before it is committed.
 */
export function formingRevisions(bar: BarFrame, revisions: number): BarFrame[] {
  if (revisions <= 0) return [bar];
  const path = barPath(bar, revisions);
  const out: BarFrame[] = [];
  let high = bar.open;
  let low = bar.open;
  const sided = bar.buy_volume !== undefined && bar.buy_volume !== null;
  for (let index = 0; index < revisions; index += 1) {
    const close = path[index];
    high = Math.max(high, close);
    low = Math.min(low, close);
    const share = (index + 1) / (revisions + 1);
    const volume = round(bar.volume * share, 4);
    const partial: Record<string, unknown> = { ...bar, high: round(high), low: round(low), close: round(close), volume, closed: false };
    if (bar.trade_count !== null) partial.trade_count = Math.max(1, Math.round(bar.trade_count * share));
    if (sided) {
      const buy = round((bar.buy_volume ?? 0) * share, 4);
      partial.buy_volume = buy;
      if (bar.sell_volume !== undefined) partial.sell_volume = round(volume - buy, 4);
    }
    out.push(partial as BarFrame);
  }
  out.push(bar);
  return out;
}

/** Every bar arriving live: `revisionsPerBar` forming revisions, then the closed bar. */
export function liveArrivals(
  bars: readonly BarFrame[],
  revisionsPerBar: number,
  tradesFor?: (bar: BarFrame, revision: number, revisions: number) => Pick<Arrival, "trades" | "tradeAvailability">,
): Arrival[] {
  const out: Arrival[] = [];
  for (const bar of bars) {
    const frames = formingRevisions(bar, revisionsPerBar);
    frames.forEach((frame, revision) => {
      out.push({ bar: frame, ...(tradesFor ? tradesFor(frame, revision, frames.length) : {}) });
    });
  }
  return out;
}

/**
 * Closed bars arriving in order except `bars[lateIndex]`, which arrives after
 * `bars[lateIndex + lateBy]`. `mergeChartBar` classifies the late bar as
 * `REBUILT`, which makes the runtime replay the whole series.
 */
export function rebuiltArrivals(bars: readonly BarFrame[], lateIndex: number, lateBy = 2): LabelledArrivals<RebuiltLabels> {
  const arrivesAfterIndex = lateIndex + lateBy;
  if (lateIndex < 0 || lateBy < 1 || arrivesAfterIndex >= bars.length) throw new Error("rebuiltArrivals: the late bar must arrive inside the series");
  const arrivals: Arrival[] = [];
  bars.forEach((bar, index) => {
    if (index === lateIndex) return;
    arrivals.push({ bar });
    if (index === arrivesAfterIndex) arrivals.push({ bar: bars[lateIndex], label: "late" });
  });
  return { arrivals, labels: { lateIndex, arrivesAfterIndex } };
}

/** `[0, step, 2·step, …]` plus the last `tail` indices: a cheaper truncation sweep for long series. */
export function sampledIndices(count: number, step: number, tail = 20): number[] {
  const out = new Set<number>();
  for (let index = 0; index < count; index += Math.max(1, step)) out.add(index);
  for (let index = Math.max(0, count - tail); index < count; index += 1) out.add(index);
  return [...out].sort((a, b) => a - b);
}

    if (!Number.isSafeInteger(capacity) || capacity <= 0) throw new Error("EventRing: capacity must be a positive integer");
    this._capacity = capacity;
  }

    return this._seq;
  }

  get synced(): boolean {
    return this._synced;
  }

    return this._lastTs;
  }

    return this._missed;

      this._events = frame.events.map(eventOf);
      this._seq = frame.seq;
      this._resetSeq = frame.seq;
      this._synced = true;
      this._missed = 0;
      this._missedThroughSeq = null;
      this._lastTs = frame.ts;
      this._trim();
      return "applied";
    }
    if (this._synced && this._seq !== null && (frame.events.length === 0 ? frame.seq < this._seq : frame.seq <= this._seq)) return "duplicate";

    if (frame.missed > 0) this._missedThroughSeq = frame.seq;
    this._lastTs = frame.ts;
    this._trim();

  }
